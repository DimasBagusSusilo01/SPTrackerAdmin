Script SQL Pembuatan Tabel Supabase

Jalankan query ini di Supabase Dashboard → SQL Editor → Run agar tabel solar_telemetry dan hak akses publik aktif:

CREATE TABLE IF NOT EXISTS public.solar_telemetry ( id BIGSERIAL PRIMARY KEY, created_at TIMESTAMPTZ DEFAULT NOW(), waktu_tanggal DATE NOT NULL DEFAULT CURRENT_DATE, waktu_jam TIME NOT NULL DEFAULT CURRENT_TIME, voltase NUMERIC(8, 2) NOT NULL, arus_listrik NUMERIC(8, 2) NOT NULL, daya NUMERIC(10, 2) NOT NULL, energi_listrik NUMERIC(10, 2) NOT NULL, irradiansi NUMERIC(8, 1) NOT NULL, besar_sudut NUMERIC(6, 1) NOT NULL, device_id TEXT DEFAULT 'SOLAR-TRACKER-01', sumber TEXT DEFAULT 'usb_serial' ); ALTER TABLE public.solar_telemetry ENABLE ROW LEVEL SECURITY; CREATE POLICY "Izinkan baca publik" ON public.solar_telemetry FOR SELECT USING (true); CREATE POLICY "Izinkan insert publik" ON public.solar_telemetry FOR INSERT WITH CHECK (true); CREATE POLICY "Izinkan delete publik" ON public.solar_telemetry FOR DELETE USING (true); DO $$ BEGIN IF NOT EXISTS ( SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND tablename = 'solar_telemetry' ) THEN ALTER PUBLICATION supabase_realtime ADD TABLE public.solar_telemetry; END IF; END $$;
